    // Carga los equipos del usuario para la sección Cuenta (y gestionar invitaciones)
    const loadUserTeamsForInvitations = async () => {
        const listElement = document.getElementById('user-teams-list');
        if (!listElement) return;
        listElement.innerHTML = 'Cargando tus equipos...';
        if (!currentUserId) return;

        try {
            const userDoc = await getDoc(doc(db, "users", currentUserId));
            const teamIds = userDoc.data()?.teams || [];

            if (teamIds.length === 0) {
                listElement.innerHTML = '<p>No eres miembro de ningún equipo.</p>';
                return;
            }

            const teamPromises = teamIds.map(id => getDoc(doc(db, "teams", id)));
            const teamDocs = await Promise.all(teamPromises);

            let html = '';
            teamDocs.forEach(teamDoc => {
                if (teamDoc.exists()) {
                    const team = teamDoc.data();
                    const membersHtml = (team.memberNames || []).map(name => `<li>${name}</li>`).join('');

                    html += `
                        <div class="equipo-card">
                            <h4>${team.name} (ID: ${teamDoc.id})</h4>
                            <p>Miembros (${team.memberNames.length}):</p>
                            <ul class="jugadores-list">${membersHtml}</ul>
                            <button data-team-id="${teamDoc.id}" data-team-name="${team.name}" class="btn-invitar">Invitar Jugador</button>
                        </div>
                    `;
                }
            });
            listElement.innerHTML = html;
            
            // Adjuntar listeners de invitación
            document.querySelectorAll('.btn-invitar').forEach(btn => {
                btn.onclick = (e) => handleInvitePlayer(e.target.dataset.teamId, e.target.dataset.teamName);

    // Carga los equipos del usuario en el selector de Crear Partido. (Lectura de cliente)
    const loadTeamsForCreation = async () => {
        const select = document.getElementById('crearPartidoConEquipo');
        if (!select) return;
        select.innerHTML = '<option value="">Selecciona tu equipo</option>';
        if (!currentUserId) return;

        try {
            const userDoc = await getDoc(doc(db, "users", currentUserId));
            const teamIds = userDoc.data()?.teams || [];

            if (teamIds.length === 0) {
                select.innerHTML = '<option value="" disabled>Debes crear un equipo primero</option>';
                return;
            }

            const teamPromises = teamIds.map(id => getDoc(doc(db, "teams", id)));
            const teamDocs = await Promise.all(teamPromises);

            teamDocs.forEach(teamDoc => {
                if (teamDoc.exists()) {
                    const team = teamDoc.data();
                    // Solo si el usuario es miembro de un equipo, lo puede seleccionar para jugar
                    if (team.members.includes(currentUserId)) {
                        const option = document.createElement('option');
                        option.value = teamDoc.id;
                        option.textContent = `${team.name} (${team.memberNames.length} miembros)`;
                        select.appendChild(option);
                    }
                }
            });

        } catch (error) {
            console.error("Error cargando equipos para creación de partido:", error);
        }
    };
